function cleanPhone(val) { if(!val) return ''; // Sirf numbers nikaalo aur agar aage +91 ya 0 laga ho toh hatakar clean 10-digit banao let num = val.toString().replace(/\D/g, ''); if(num.length > 10 && num.startsWith('91')) num = num.slice(2); if(num.length > 10 && num.startsWith('0')) num = num.slice(1); return num; } function handleForgotStep1(e) { e.preventDefault(); const rawId = document.getElementById('forgotId').value.trim(); const cleanedId = cleanPhone(rawId); const msg = document.getElementById('forgotMsg'); // Taaza data fetch karo localStorage se try { users = JSON.parse(localStorage.getItem('pag_users_db')) || []; } catch(err) { users = []; } // Mobile number ya email dono check karo const user = users.find(u => { const userCleanPhone = cleanPhone(u.phone); return (userCleanPhone && userCleanPhone === cleanedId) || (u.email && u.email.toLowerCase() === rawId.toLowerCase()); }); if(!user) { msg.className = "auth-msg err"; msg.innerHTML = `Account not found for ${rawId}!
Pehle Sign Up karein ya WhatsApp Admin se madad lein.`; return; } if(!user.secQuestion || !user.secAnswer) { msg.className = "auth-msg err"; msg.innerHTML = `Is account par security question set nahi hai.
WhatsApp par Direct Reset lein.`; return; } userVerifying = user; msg.className = "auth-msg ok"; msg.innerText = `Account mil gaya: ${user.name}! Niche apna security answer bharein.`; document.getElementById('displaySecQuestion').innerText = user.secQuestion; document.getElementById('forgotStep1Form').style.display = 'none'; document.getElementById('forgotStep2Box').style.display = 'block'; }
